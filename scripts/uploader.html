const FOLDER_ID = "1hu1kGOqDz2zM8H7aNd4SzMQl9ru3FB7n";
const SECRET_KEY = "anishkumwat";

function doGet(e) {
  try {
    const props = PropertiesService.getScriptProperties();
    let notif = null;
    const raw = props.getProperty("notification");
    if (raw) {
      notif = JSON.parse(raw);
    } else {
      const folder = DriveApp.getFolderById(FOLDER_ID);
      const files = folder.getFilesByName("moonlight-notification.json");
      if (files.hasNext()) {
        notif = JSON.parse(files.next().getBlob().getDataAsString());
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ ok: true, notification: notif }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    if (!data.key || data.key !== SECRET_KEY) {
      return ContentService.createTextOutput(JSON.stringify({ ok: false, error: "Wrong secret key" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    const folder = DriveApp.getFolderById(FOLDER_ID);
    const props = PropertiesService.getScriptProperties();

    // 1. DELETE SONG & MATCHING THUMBNAIL
    if (data.action === "delete_song") {
      const fileId = data.songId;
      let audioStem = "";
      try {
        const audioFile = DriveApp.getFileById(fileId);
        audioStem = audioFile.getName().replace(/\.[^.]+$/, "").trim();
        audioFile.setTrashed(true);
      } catch (err) {}

      // Trash matching cover file(s)
      const allFiles = folder.getFiles();
      while (allFiles.hasNext()) {
        const f = allFiles.next();
        const fname = f.getName();
        const isCover = fname.startsWith("moonlight-cover-") || /\.(jpe?g|png)$/i.test(fname);
        if (fname.includes(fileId) || (audioStem && isCover && fname.toLowerCase().startsWith(audioStem.toLowerCase()))) {
          f.setTrashed(true);
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ ok: true, deleted: fileId }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // 2. UPDATE SONG (Title, Artist, Categories, or Cover)
    if (data.action === "update_song") {
      const fileId = data.songId;
      const audioFile = DriveApp.getFileById(fileId);
      if (data.newTitle) {
        audioFile.setName(data.newTitle + ".mp3");
      }

      // If Cloudinary cover URL provided, set file description
      if (data.newCoverUrl) {
        audioFile.setDescription(data.newCoverUrl);
      }

      // If new thumbnail is uploaded
      let newCoverId = null;
      if (data.newThumb) {
        // Trash old cover
        const allFiles = folder.getFiles();
        while (allFiles.hasNext()) {
          const f = allFiles.next();
          if (f.getName().startsWith("moonlight-cover-" + fileId)) {
            f.setTrashed(true);
          }
        }
        const ext = data.thumbExt || "jpg";
        const thumbBlob = Utilities.newBlob(
          Utilities.base64Decode(data.newThumb),
          data.thumbType || "image/jpeg",
          "moonlight-cover-" + fileId + "." + ext
        );
        const newThumbFile = folder.createFile(thumbBlob);
        newThumbFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        newCoverId = newThumbFile.getId();
      }

      return ContentService.createTextOutput(JSON.stringify({
        ok: true,
        updated: fileId,
        title: data.newTitle,
        coverId: newCoverId
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // 3. SET PUSH NOTIFICATION
    if (data.action === "set_notification") {
      const notifData = {
        id: data.id || ("n_" + Date.now()),
        title: data.title || "Hey {name}! 🌙",
        tagline: data.tagline || "",
        imageUrl: data.imageUrl || "",
        songId: data.songId || "",
        active: data.active !== false,
        createdAt: Date.now()
      };
      props.setProperty("notification", JSON.stringify(notifData));

      const files = folder.getFilesByName("moonlight-notification.json");
      if (files.hasNext()) {
        files.next().setContent(JSON.stringify(notifData));
      } else {
        folder.createFile("moonlight-notification.json", JSON.stringify(notifData), MimeType.PLAIN_TEXT);
      }

      return ContentService.createTextOutput(JSON.stringify({ ok: true, notification: notifData }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // 4. CLEAR PUSH NOTIFICATION
    if (data.action === "clear_notification") {
      props.deleteProperty("notification");
      const files = folder.getFilesByName("moonlight-notification.json");
      while (files.hasNext()) files.next().setTrashed(true);
      return ContentService.createTextOutput(JSON.stringify({ ok: true, cleared: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // 5. UPLOAD NEW SONG (MP3 + Cover)
    if (data.audio) {
      const audioBlob = Utilities.newBlob(Utilities.base64Decode(data.audio), "audio/mpeg", (data.title || "Untitled") + ".mp3");
      const audioFile = folder.createFile(audioBlob);
      audioFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

      // Save Cloudinary cover URL in Drive file description (instant CDN for app)
      if (data.coverUrl) {
        audioFile.setDescription(data.coverUrl);
      }

      let thumbFile = null;
      if (data.thumb) {
        const ext = data.thumbExt || "jpg";
        const thumbBlob = Utilities.newBlob(
          Utilities.base64Decode(data.thumb),
          data.thumbType || "image/jpeg",
          "moonlight-cover-" + audioFile.getId() + "." + ext
        );
        thumbFile = folder.createFile(thumbBlob);
        thumbFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      }

      const res = {
        ok: true,
        id: audioFile.getId(),
        title: data.title,
        audioName: audioFile.getName(),
        thumbName: thumbFile ? thumbFile.getName() : "",
        mp3Link: "https://drive.google.com/uc?export=download&id=" + audioFile.getId(),
        thumbLink: thumbFile ? "https://drive.google.com/uc?export=view&id=" + thumbFile.getId() : "",
        thumbPreviewLink: thumbFile ? "https://drive.google.com/thumbnail?id=" + thumbFile.getId() + "&sz=w400" : ""
      };

      return ContentService.createTextOutput(JSON.stringify(res))
        .setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: "Invalid action or missing data" }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
